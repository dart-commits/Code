        else high = mid - 1;

void swap(int *a, int *b) { int t = *a; *a = *b; *b = t; }
int partition(int a[], int low, int high) {
    int pivot = a[high], i = low - 1, j;
    for (j = low; j < high; j++)
        if (a[j] <= pivot) { i++; swap(&a[i], &a[j]); }
    swap(&a[i + 1], &a[high]);
    return i + 1;
}
void quickSort(int a[], int low, int high) {
    if (low < high) {
        int p = partition(a, low, high);
        quickSort(a, low, p - 1);
        quickSort(a, p + 1, high);
    }
}

#include <stdio.h>
void merge(int a[], int l, int m, int r) {
    int n1 = m - l + 1, n2 = r - m, i, j, k;
    int L[n1], R[n2];
    for (i = 0; i < n1; i++) L[i] = a[l + i];
    for (j = 0; j < n2; j++) R[j] = a[m + 1 + j];
    i = 0; j = 0; k = l;
    while (i < n1 && j < n2)
        a[k++] = (L[i] <= R[j]) ? L[i++] : R[j++];
    while (i < n1) a[k++] = L[i++];
    while (j < n2) a[k++] = R[j++];
}
void mergeSort(int a[], int l, int r) {
    if (l < r) {
        int m = l + (r - l) / 2;
        mergeSort(a, l, m); mergeSort(a, m + 1, r);
        merge(a, l, m, r);
    }
}

#include <stdio.h>
#include <stdlib.h>
typedef struct Node { int data; struct Node *next; } Node;
void insertEnd(Node **head, int x) {
    Node *p = malloc(sizeof(Node)), *t;
    if (!p) return;
    p->data = x; p->next = NULL;
    if (*head == NULL) { *head = p; return; }
    for (t = *head; t->next; t = t->next) {}
    t->next = p;
}
void display(Node *p) {
    while (p) { printf("%d -> ", p->data); p = p->next; }
    puts("NULL");
}
void deleteValue(Node **head, int x) {
    Node *p = *head, *prev = NULL;
    while (p && p->data != x) { prev = p; p = p->next; }
    if (!p) { puts("Value not found"); return; }
    if (prev) prev->next = p->next; else *head = p->next;
    free(p);
}
int main(void) {

12. INFIX TO POSTFIX (SINGLE-CHARACTER OPERANDS)
------------------------------------------------
#include <stdio.h>
#include <ctype.h>
#define MAX 100
char st[MAX]; int top = -1;
void push(char c) { st[++top] = c; }
char pop(void) { return st[top--]; }
int prec(char c) {
    if (c == '^') return 3;
    if (c == '*' || c == '/') return 2;
    if (c == '+' || c == '-') return 1;
    return 0;
}
int main(void) {
    char in[MAX], out[MAX]; int i, k = 0; char c;
    printf("Enter infix (no spaces): "); scanf("%99s", in);
    for (i = 0; in[i]; i++) {
        c = in[i];
        if (isalnum((unsigned char)c)) out[k++] = c;
        else if (c == '(') push(c);
        else if (c == ')') {
            while (top >= 0 && st[top] != '(') out[k++] = pop();
            if (top >= 0) pop();
        } else {
            while (top >= 0 && st[top] != '(' &&
                   (prec(st[top]) > prec(c) ||
                   (prec(st[top]) == prec(c) && c != '^')))
                out[k++] = pop();
            push(c);
        }
    }
    while (top >= 0) out[k++] = pop();
    out[k] = '\0';
    printf("Postfix: %s\n", out);
    return 0;
}

16. MAX HEAP — CREATION, INSERTION, DELETION (EXTRACT MAX)

#include <stdio.h>
#define MAX 100
int h[MAX], size = 0;
void swap(int *a, int *b) { int t = *a; *a = *b; *b = t; }
void siftDown(int i) {
    while (1) {
        int largest = i, l = 2*i + 1, r = 2*i + 2;
        if (l < size && h[l] > h[largest]) largest = l;
        if (r < size && h[r] > h[largest]) largest = r;
        if (largest == i) break;
        swap(&h[i], &h[largest]); i = largest;
    }
}
void insert(int x) {
    int i;
    if (size == MAX) { puts("Heap full"); return; }
    i = size++; h[i] = x;
    while (i > 0 && h[(i-1)/2] < h[i]) {
        swap(&h[i], &h[(i-1)/2]); i = (i-1)/2;
    }
}
int extractMax(void) {
    int result;
    if (!size) { puts("Heap empty"); return -1; }
    result = h[0]; h[0] = h[--size]; if (size) siftDown(0);
    return result;
}
void buildHeap(void) { int i; for (i = size/2 - 1; i >= 0; i--) siftDown(i); }
void display(void) { int i; for (i = 0; i < size; i++) printf("%d ", h[i]); puts(""); }
int main(void) {
    int n, i, x, ch;
    printf("Initial number of elements: "); scanf("%d", &n);
    for (i = 0; i < n && i < MAX; i++) scanf("%d", &h[i]);

    do {
        printf("1.Insert 2.Delete max 3.Display 4.Exit: "); scanf("%d", &ch);
        if (ch == 1) { scanf("%d", &x); insert(x); }
        else if (ch == 2) printf("Deleted max: %d\n", extractMax());

}

17. HEAP SORT (ASCENDING)

void heapify(int a[], int n, int i) {
    int largest = i, l = 2*i + 1, r = 2*i + 2;
    if (l < n && a[l] > a[largest]) largest = l;
    if (r < n && a[r] > a[largest]) largest = r;
    if (largest != i) { swap(&a[i], &a[largest]); heapify(a, n, largest); }
}
void heapSort(int a[], int n) {
    int i;
    for (i = n/2 - 1; i >= 0; i--) heapify(a, n, i);

        swap(&a[0], &a[i]); heapify(a, i, 0);
    }
}

void DFS(int u) {
    int v;
    visited[u] = 1;
    printf("%d ", u);
    for (v = 0; v < n; v++)
        if (a[u][v] && !visited[v]) DFS(v);
}
int main(void) {
    int i, j, start;
    printf("Number of vertices (1-%d): ", MAX); if (scanf("%d", &n) != 1 || n < 1 || n > MAX) { puts("Invalid vertex count"); return 1; }
    printf("Enter adjacency matrix:\n");
    for (i = 0; i < n; i++)
        for (j = 0; j < n; j++) scanf("%d", &a[i][j]);
    printf("Starting vertex (0 to %d): ", n - 1); if (scanf("%d", &start) != 1 || start < 0 || start >= n) { puts("Invalid start vertex"); return 1; }
    printf("DFS: "); DFS(start); puts("");
    return 0;
}

21. PRIM'S ALGORITHM (MINIMUM SPANNING TREE)
--------------------------------------------
#include <stdio.h>
#define MAX 100
#define INF 1000000000
int main(void) {
    int n, g[MAX][MAX], selected[MAX] = {0}, i, j, edges = 0;
    int total = 0;
    printf("Number of vertices (1-%d): ", MAX); if (scanf("%d", &n) != 1 || n < 1 || n > MAX) { puts("Invalid vertex count"); return 1; }
    printf("Enter weight matrix (0 means no edge; diagonal 0):\n");
    for (i = 0; i < n; i++)
        for (j = 0; j < n; j++) scanf("%d", &g[i][j]);
    selected[0] = 1;
    puts("Edge  Weight");
    while (edges < n - 1) {
        int min = INF, u = -1, v = -1;
        for (i = 0; i < n; i++) if (selected[i])
            for (j = 0; j < n; j++)
                if (!selected[j] && g[i][j] > 0 && g[i][j] < min) {
                    min = g[i][j]; u = i; v = j;
                }
        if (u == -1) { puts("Graph is disconnected; MST not possible."); return 0; }
        printf("%d - %d  %d\n", u, v, min);
        selected[v] = 1; total += min; edges++;
    }
    printf("Total MST weight: %d\n", total);
    return 0;
}

22. FLOYD–WARSHALL (ALL-PAIRS SHORTEST PATHS)
--------------------------------------------
#include <stdio.h>
#define MAX 100
#define INF 1000000
int main(void) {
    int n, d[MAX][MAX], i, j, k;
    printf("Number of vertices (1-%d): ", MAX); if (scanf("%d", &n) != 1 || n < 1 || n > MAX) { puts("Invalid vertex count"); return 1; }
    printf("Enter weight matrix (%d means no edge; diagonal 0):\n", INF);
    for (i = 0; i < n; i++)
        for (j = 0; j < n; j++) scanf("%d", &d[i][j]);
    for (k = 0; k < n; k++)
        for (i = 0; i < n; i++)
            for (j = 0; j < n; j++)
                if (d[i][k] != INF && d[k][j] != INF &&
                    d[i][j] > d[i][k] + d[k][j])
                    d[i][j] = d[i][k] + d[k][j];
    puts("All-pairs shortest path matrix:");
    for (i = 0; i < n; i++) {
        for (j = 0; j < n; j++)
            if (d[i][j] == INF) printf("INF ");
            else printf("%d ", d[i][j]);
        puts("");
    }

}

23. HASH TABLE — LINEAR PROBING (INSERT, SEARCH, DELETE, DISPLAY)
-----------------------------------------------------------------
#include <stdio.h>
#define SIZE 10
#define EMPTY -1
#define DELETED -2
int table[SIZE];
int hash(int key) { int h = key % SIZE; return h < 0 ? h + SIZE : h; }
void init(void) { int i; for (i = 0; i < SIZE; i++) table[i] = EMPTY; }
void insert(int key) {
    int i, idx = hash(key);
    for (i = 0; i < SIZE; i++) {
        int p = (idx + i) % SIZE;
        if (table[p] == EMPTY || table[p] == DELETED) {
            table[p] = key; printf("Inserted at index %d\n", p); return;
        }
        if (table[p] == key) { puts("Duplicate key"); return; }
    }
    puts("Hash table full");
}
int search(int key) {
    int i, idx = hash(key);
    for (i = 0; i < SIZE; i++) {
        int p = (idx + i) % SIZE;
        if (table[p] == EMPTY) return -1;
        if (table[p] == key) return p;
    }
    return -1;
}
void deleteKey(int key) {
    int p = search(key);
    if (p < 0) puts("Key not found");
    else { table[p] = DELETED; puts("Deleted"); }
}
void display(void) {
    int i;
    for (i = 0; i < SIZE; i++) {
        printf("%d: ", i);
        if (table[i] == EMPTY) puts("EMPTY");
        else if (table[i] == DELETED) puts("DELETED");
        else printf("%d\n", table[i]);
    }
}
int main(void) {
    int ch, key; init();
    do {
        printf("1.Insert 2.Search 3.Delete 4.Display 5.Exit: "); scanf("%d", &ch);
        if (ch == 1) { scanf("%d", &key); insert(key); }
        else if (ch == 2) { scanf("%d", &key); printf("%s\n", search(key) < 0 ? "Not found" : "Found"); }
        else if (ch == 3) { scanf("%d", &key); deleteKey(key); }
        else if (ch == 4) display();
    } while (ch != 5);
    return 0;
}

24. HASH TABLE — SEPARATE CHAINING (INSERT, SEARCH, DELETE, DISPLAY)
-------------------------------------------------------------------
#include <stdio.h>
#include <stdlib.h>
#define SIZE 10
typedef struct Node { int key; struct Node *next; } Node;
Node *table[SIZE] = {NULL};
int hash(int key) { int h = key % SIZE; return h < 0 ? h + SIZE : h; }
void insert(int key) {
    int i = hash(key); Node *p;
    for (p = table[i]; p; p = p->next)
        if (p->key == key) { puts("Duplicate key"); return; }
    p = malloc(sizeof(Node)); if (!p) return;
    p->key = key; p->next = table[i]; table[i] = p;
}
int search(int key) {
    Node *p;
    for (p = table[hash(key)]; p; p = p->next)
        if (p->key == key) return 1;
    return 0;
}
void deleteKey(int key) {
    int i = hash(key); Node *p = table[i], *prev = NULL;
    while (p && p->key != key) { prev = p; p = p->next; }
    if (!p) { puts("Key not found"); return; }
    if (prev) prev->next = p->next; else table[i] = p->next;
    free(p); puts("Deleted");
}
void display(void) {
    int i; Node *p;
    for (i = 0; i < SIZE; i++) {
        printf("%d: ", i);
        for (p = table[i]; p; p = p->next) printf("%d -> ", p->key);
        puts("NULL");
    }
}
int main(void) {
    int ch, key;
    do {
        printf("1.Insert 2.Search 3.Delete 4.Display 5.Exit: "); scanf("%d", &ch);
        if (ch == 1) { scanf("%d", &key); insert(key); }
        else if (ch == 2) { scanf("%d", &key); puts(search(key) ? "Found" : "Not found"); }
        else if (ch == 3) { scanf("%d", &key); deleteKey(key); }
        else if (ch == 4) display();
    } while (ch != 5);
    return 0;
}
